Η Μεγαλειώδης Υπόθεσις
των Ευγενών Φασολίων

✦ ❧ ✦

Εν ᾗ ο τλήμων Μαθητής, μεταμορφωθείς εις Έντομον, καλείται να συλλέξη Ζουμάκιον εκ των μη δεδηλητηριασμένων Πιάτων — αψηφών τας ραδιουργίας του κυρίου Νετανιάχου και της τρομεράς Γιαγιάς.

⚜ Αλγόριθμος: Dijkstra + Bitmask DP ⚜
📜 Εκφώνησις & Μέθοδος

Δίδεται βεβαρημένος γράφος N κόμβων (ήτοι πιάτων φασολίων). Ο αλεξίκακος κύριος Νετανιάχου, εν συνεργασία μετά της πανούργου Γιαγιάς, ηλλοίωσε τα K πιάτα με το ολιγώτερον λάδι — αυτά είναι δεδηλητηριασμένα. Ο Μαθητής ξεκινά εκ κόμβου 0 και οφείλει να επισκεφθή άπαντα τα ασφαλή πιάτα με ελαχίστη ολική απόστασιν. Επιτρέπεται η διέλευσις εκ δεδηλητηριασμένων κόμβων — πλην ου πίνει!

Ⅰ
Εντοπισμός Δεδηλητηριασμένων: Ταξινόμησε τους κόμβους κατ' ανερχόμενον λάδι (sort). Σημείωσε τους K πρώτους ως poisoned[i] = true. Οι λοιποί N−K είναι ασφαλείς.
Ⅱ
Dijkstra εκ πάντων των Ασφαλών + Αφετηρίας: Τρέξε dijkstra(s) εκ κάθε ασφαλούς κόμβου και εκ του κόμβου 0. Αποθήκευσε dist[i][j]. Τα ενδιάμεσα δηλητηριασμένα πιάτα παρακάμπτονται αφανώς.
Ⅲ
Bitmask DP — Περιοδεία Κόμβων: Ορισμός: dp[mask][i] = ελάχιστον κόστος δια την επίσκεψιν των κόμβων τού mask, τελευταίον εκ των οποίων ήτο ο i.
Βάσις: dp[1<<i][i] = dist[0][safe[i]]
Μετάβασις: dp[mask|(1<<j)][j] = min(dp[mask][i] + dist[i][j])
Ⅳ
Απάντησις: min{ dp[FULL][i] } όπου FULL = (1<<S)−1 και S ο αριθμός ασφαλών πιάτων.
Dijkstra: O(S × (N+M) log N) Bitmask DP: O(2^S × S²) Σύνολον: O(S·(N+M)logN + 2^S·S²) ⚠ Χρησιμοποίει long long — οι αποστάσεις μπορούν να εκτοξευθούν! ✓ Αντέχει S ≤ 20, N ≤ 10⁵ 
// ═══════════════════════════════════════════════════
// Τα Φασόλια της Γιαγιάς — Βασιλική Λύσις
// Dijkstra + Bitmask DP   (0-indexed)
// ═══════════════════════════════════════════════════
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<ll,int> pli;

const ll INF = 1e18;
const int MAXN = 305;

int N, M, K;
vector<pair<int,ll>> adj[MAXN];
ll   oil[MAXN];
bool poisoned[MAXN];
int  safe_id[20];       // ασφαλείς κόμβοι
ll   D[22][MAXN];       // D[i] = dist από safe_id[i], D[S] = από κόμβο 0

void dijkstra(int src, ll* d) {
    fill(d, d+N, INF);
    d[src] = 0;
    priority_queue<pli, vector<pli>, greater<pli>> pq;
    pq.push({0, src});
    while (!pq.empty()) {
        auto [du, u] = pq.top(); pq.pop();
        if (du > d[u]) continue;
        for (auto [v, w] : adj[u])
            if (d[u]+w < d[v]) {
                d[v] = d[u]+w;
                pq.push({d[v], v});
            }
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> N >> M;

    vector<pair<ll,int>> by_oil(N);
    for (int i = 0; i < N; i++) {
        cin >> oil[i];
        by_oil[i] = {oil[i], i};
    }

    for (int i = 0; i < M; i++) {
        int u, v; ll w;
        cin >> u >> v >> w;
        adj[u].push_back({v, w});
        adj[v].push_back({u, w});
    }

    cin >> K;

    // Τα K πιάτα με ολιγώτατον λάδι = δεδηλητηριασμένα
    sort(by_oil.begin(), by_oil.end());
    for (int i = 0; i < K; i++)
        poisoned[by_oil[i].second] = true;

    int S = 0;
    for (int i = 0; i < N; i++)
        if (!poisoned[i]) safe_id[S++] = i;

    // Dijkstra από κάθε ασφαλή + αφετηρία 0
    for (int i = 0; i < S; i++) dijkstra(safe_id[i], D[i]);
    dijkstra(0, D[S]); // D[S] = από κόμβο 0

    // Bitmask DP
    int FULL = (1<<S)-1;
    vector<vector<ll>> dp(1<<S, vector<ll>(S, INF));

    for (int i = 0; i < S; i++)
        dp[1<<i][i] = D[S][safe_id[i]]; // βάσις

    for (int mask = 1; mask <= FULL; mask++)
        for (int i = 0; i < S; i++) {
            if (!(mask&(1<<i)) || dp[mask][i]==INF) continue;
            for (int j = 0; j < S; j++) {
                if (mask&(1<<j)) continue;
                ll nd = dp[mask][i] + D[i][safe_id[j]];
                dp[mask|(1<<j)][j] = min(dp[mask|(1<<j)][j], nd);
            }
        }

    ll ans = INF;
    for (int i = 0; i < S; i++)
        ans = min(ans, dp[FULL][i]);

    cout << ans << "\n";
    return 0;
}
