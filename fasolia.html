Τα Φασόλια της Γιαγιάς

Dijkstra + Bitmask DP

Ο μαθητής (σε μέγεθος εντόμου) πρέπει να επισκεφτεί όλα τα ασφαλή πιάτα με ελάχιστη συνολική απόσταση. Τα K πιάτα με το λιγότερο λάδι είναι δηλητηριασμένα — μπορεί να τα διασχίσει αλλά δεν πίνει από αυτά.

Εκφώνηση & Αλγόριθμος
Δίνεται weighted graph με N πιάτα (κόμβοι). Τα K πιάτα με το λιγότερο λάδι είναι δηλητηριασμένα. Ο μαθητής ξεκινά από τον κόμβο 0 και πρέπει να επισκεφτεί όλα τα ασφαλή πιάτα με ελάχιστη συνολική απόσταση. Μπορεί να περνά από δηλητηριασμένους κόμβους — απλά δεν πίνει.

(ο κόμβος 0 ΔΕΝ χρειάζεται επίσκεψη, είναι αφετηρία — εξαρτάται από την εκφώνηση)
1
Βρες τα δηλητηριασμένα πιάτα: Ταξινόμησε τους κόμβους κατά λάδι (sort) και σημείωσε τους K με το λιγότερο ως poisoned[]. Οι υπόλοιποι N-K είναι ασφαλείς.
2
Dijkstra από κάθε ασφαλή + αφετηρία: Τρέξε Dijkstra(s) από κάθε ασφαλή κόμβο και τον κόμβο 0. Έτσι βρίσκεις τις ελάχιστες αποστάσεις μεταξύ τους — ακόμα και αν το μονοπάτι περνά από δηλητηριασμένους. Αποθήκευσε dist[i][j].
3
Bitmask DP: Ορισμός: dp[mask][i] = ελάχιστο κόστος για να έχεις επισκεφτεί τους ασφαλείς κόμβους που δηλώνει το mask, με τελευταίο τον i.
Βάση: dp[1<<i][i] = dist[start][safe[i]]
Μετάβαση: dp[mask|(1<<j)][j] = min(dp[mask][i] + dist[i][j])
4
Απάντηση: min over i of dp[(1<<S)-1][i] όπου S = N-K (αριθμός ασφαλών).
Πολυπλοκότητα: Dijkstra: O(S × (N+M) log N) Bitmask DP: O(2^S × S²) Σύνολο: O(S·(N+M)logN + 2^S·S²) Όρια: Δουλεύει για S ≤ 20 (bitmask), N ≤ 10^5 (Dijkstra) Προσοχή: Χρησιμοποίησε long long για τα βάρη! 
// Τα Φασόλια της Γιαγιάς — Dijkstra + Bitmask DP
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef pair<ll,int> pli;

const ll INF = 1e18;
const int MAXN = 305;

int N, M, K;
vector<pair<int,ll>> adj[MAXN];
ll   oil[MAXN];
bool poisoned[MAXN];
int  safe_id[20]; // indices of safe nodes
ll   dist_from[22][MAXN]; // [0..S-1] από safe nodes, [S] από κόμβο 0

void dijkstra(int src, ll* d) {
    fill(d, d+N, INF);
    d[src] = 0;
    priority_queue<pli, vector<pli>, greater<pli>> pq;
    pq.push({0, src});
    while (!pq.empty()) {
        auto [du, u] = pq.top(); pq.pop();
        if (du > d[u]) continue;
        for (auto [v, w] : adj[u]) {
            if (d[u] + w < d[v]) {
                d[v] = d[u] + w;
                pq.push({d[v], v});
            }
        }
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> N >> M;

    vector<pair<ll,int>> oil_sorted(N);
    for (int i = 0; i < N; i++) {
        cin >> oil[i];
        oil_sorted[i] = {oil[i], i};
    }

    for (int i = 0; i < M; i++) {
        int u, v; ll w;
        cin >> u >> v >> w;
        adj[u].push_back({v, w});
        adj[v].push_back({u, w}); // undirected
    }

    cin >> K;

    // Βρες τα K πιάτα με λιγότερο λάδι = δηλητηριασμένα
    sort(oil_sorted.begin(), oil_sorted.end());
    for (int i = 0; i < K; i++)
        poisoned[oil_sorted[i].second] = true;

    int S = 0;
    for (int i = 0; i < N; i++)
        if (!poisoned[i]) safe_id[S++] = i;

    // Dijkstra από κάθε ασφαλή κόμβο
    for (int i = 0; i < S; i++)
        dijkstra(safe_id[i], dist_from[i]);
    // dist_from[S] = από κόμβο 0 (αφετηρία)
    dijkstra(0, dist_from[S]);

    // Bitmask DP
    int FULL = (1 << S) - 1;
    vector<vector<ll>> dp(1 << S, vector<ll>(S, INF));

    // Βάση: ξεκίνα από κόμβο 0, πήγαινε στον safe[i]
    for (int i = 0; i < S; i++)
        dp[1<<i][i] = dist_from[S][safe_id[i]];

    for (int mask = 1; mask <= FULL; mask++) {
        for (int i = 0; i < S; i++) {
            if (!(mask & (1<<i))) continue;
            if (dp[mask][i] == INF) continue;
            for (int j = 0; j < S; j++) {
                if (mask & (1<<j)) continue;
                int  nm = mask | (1<<j);
                ll   nd = dp[mask][i] + dist_from[i][safe_id[j]];
                dp[nm][j] = min(dp[nm][j], nd);
            }
        }
    }

    ll ans = INF;
    for (int i = 0; i < S; i++)
        ans = min(ans, dp[FULL][i]);

    cout << ans << "\n";
    return 0;
}
